91/// history/archive 共用写真リスト HTML 生成。
92///
93/// * `pic_list` - 画像データ。
94/// * `start` - pic_list の何番目から表示するか。
95/// * `page_by` - start からいくつ画像を表示するか。
96/// * `title` - タイトル。
97/// * `commands` - POST の "cmd" パラメータで送られる値とラジオボタンに
98///   添えるラベルからなるタプルの配列
99fn create_pic_list_page(
100    pic_list: &BTreeMap<String, PicEntry>,
101    img_path_dir: &str,
102    start: usize,
103    page_by: usize,
104    title: &str,
105    commands: &[(&str, &str)],
106) -> String {
107    let total = pic_list.len();
108    let data: Vec<_> = pic_list.keys().rev().skip(start).take(page_by).collect();
109
110    let mut fig_list = String::new();
111    fig_list += r#"    <form method="post">
112      <fieldset>
113        <legend>Commands for selected items</legend>
114        <input type="submit" value="Execute">
115"#;
116    let mut is_first = true;
117    for &(cmd, label) in commands {
118        let checked = if is_first {
119            is_first = false;
120            r#" checked="checked""#
121        } else {
122            ""
123        };
124        fig_list += &format!(
125            r#"        <label><input type="radio" name="cmd" value="{cmd}"{checked}>{label}</label>
126"#
127        );
128    }
129    fig_list += r#"      </fieldset>
130      <p><input type="reset" value="Reset"></p>
131
132"#;
133
134    for name in data {
135        fig_list += &format!(
136            r#"      <input type="checkbox" id="{name}" name="target" value="{name}">
137      <figure class="pic">
138        <a href="./pic/{img_path_dir}/{name}/main"><img src="./pic/{img_path_dir}/{name}/thumb" alt="{name}"></a>
139        <figcaption class="pic"><label for="{name}">{name}</label></figcaption>
140      </figure>
141"#
142        );
143    }
144    fig_list += "    </form>";
145
146    let mut page_navi = String::new();
147    page_navi += &format!("<p>{} files</p>", pic_list.len());
148    page_navi += "<p>";
149    for left in (0..total).step_by(page_by) {
150        let right = cmp::min(left + page_by - 1, total - 1);
151        let link = if (left..=right).contains(&start) {
152            format!(r#"{}-{} "#, left + 1, right + 1)
153        } else {
154            format!(
155                r#"<a href="?page={}">{}-{}</a> "#,
156                left,
157                left + 1,
158                right + 1
159            )
160        };
161        page_navi += &link;
162    }
163    page_navi += "</p>";
164
165    format!(
166        r#"<!DOCTYPE html>
167<html lang="en">
168  <head>
169    <title>{title}</title>
170    <style>
171      figure.pic {{
172        display: inline-block;
173        margin: 5px;
174      }}
175      figcaption.pic {{
176        font-size: 100%;
177        text-align: center;
178      }}
179    </style>
180  </head>
181  <body>
182    <h1>{title}</h1>
183      {page_navi}
184
185{fig_list}
186
187    <h2>Navigation</h2>
188    <p><a href="./">Camera Main Page</a></p>
189  </body>
190</html>
191"#
192    )
193}

195/// 同一キーはデシリアライズ対応していないので自力でパースする。
196///
197/// cmd=c&target=t1&target=t2&...
198fn parse_cmd_targets(
199    iter: impl IntoIterator<Item = (String, String)>,
200) -> Result<(String, Vec<String>)> {
201    let mut cmd = None;
202    let mut targets = Vec::new();
203
204    for (key, value) in iter {
205        match key.as_str() {
206            "cmd" => {
207                cmd = Some(value);
208            }
209            "target" => {
210                targets.push(value);
211            }
212            _ => {}
213        }
214    }
215
216    cmd.map_or_else(|| Err(anyhow!("cmd is required")), |cmd| Ok((cmd, targets)))
217}

219/// POST /priv/camera/history
220///
221/// * `cmd` - "archive" or "delete"
222/// * `target` - 対象の picture ID (複数回指定可)
223#[actix_web::post("/camera/history")]
224async fn history_post(
225    ctrl: web::Data<Control>,
226    form: web::Form<Vec<(String, String)>>,
227) -> HttpResponse {
228    let param = parse_cmd_targets(form.0);
229    if let Err(why) = param {
230        return HttpResponse::BadRequest()
231            .content_type(ContentType::plaintext())
232            .body(why.to_string());
233    }
234    let (cmd, targets) = param.unwrap();
235
236    match cmd.as_str() {
237        "archive" => {
238            if let Err(why) = archive_pics(&ctrl, &targets).await {
239                if why.is::<std::io::Error>() {
240                    HttpResponse::InternalServerError()
241                        .content_type(ContentType::plaintext())
242                        .body(why.to_string())
243                } else {
244                    HttpResponse::BadRequest()
245                        .content_type(ContentType::plaintext())
246                        .body(why.to_string())
247                }
248            } else {
249                // 成功したら archive GET へリダイレクト
250                HttpResponse::SeeOther()
251                    .append_header(("LOCATION", "./archive"))
252                    .finish()
253            }
254        }
255        "delete" => {
256            if let Err(why) = delete_history_pics(&ctrl, &targets).await {
257                HttpResponse::BadRequest()
258                    .content_type(ContentType::plaintext())
259                    .body(why.to_string())
260            } else {
261                // 成功したら history GET へリダイレクト
262                HttpResponse::SeeOther()
263                    .append_header(("LOCATION", "./history"))
264                    .finish()
265            }
266        }
267        _ => HttpResponse::BadRequest()
268            .content_type(ContentType::plaintext())
269            .body("invalid command"),
270    }
271}

361async fn pic_get_internal(ctrl: &Control, stype: StorageType, name: &str, kind: &str) -> WebResult {
362    let is_th = match kind {
363        "main" => false,
364        "thumb" => true,
365        _ => return Ok(error_resp(StatusCode::BAD_REQUEST)),
366    };
367
368    let value = {
369        let camera = ctrl.sysmods().camera.lock().await;
370        let (hist, ar) = camera.pic_list();
371        let dict = match stype {
372            StorageType::History => hist,
373            StorageType::Archive => ar,
374        };
375        dict.get(name).cloned()
376    };
377
378    if let Some(entry) = value {
379        let path = match is_th {
380            false => entry.path_main,
381            true => entry.path_th,
382        };
383        let mut file = File::open(path).await.map_err(|e| anyhow!(e))?;
384        let mut bin = Vec::new();
385        let _ = file.read_to_end(&mut bin).await.map_err(|e| anyhow!(e))?;
386
387        let resp = HttpResponse::Ok()
388            .content_type(ContentType::jpeg())
389            .body(bin);
390        Ok(resp)
391    } else {
392        Ok(error_resp(StatusCode::NOT_FOUND))
393    }
394}

396#[actix_web::post("/camera/take")]
397async fn take_post(ctrl: web::Data<Control>) -> WebResult {
398    let pic = take_a_pic(TakePicOption::new()).await;
399    if let Err(ref e) = pic {
400        error!("take a picture error");
401        error!("{e:#}");
402    }
403    let pic = pic?;
404
405    let thumb = create_thumbnail(&pic)?;
406
407    let mut camera = ctrl.sysmods().camera.lock().await;
408    camera.push_pic_history(&pic, &thumb).await?;
409    drop(camera);
410
411    let resp = HttpResponse::Ok()
412        .content_type(ContentType::jpeg())
413        .body(pic);
414    Ok(resp)
